import React, { useState, useEffect } from "react";

export const DeepLinkAccordion = ({ title, children }) => {
  const slugify = (text) =>
    (text || "section")
      .toLowerCase()
      .trim()
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/^-+|-+$/g, "") || "section";

  const AnchorIcon = () => (
    <svg
      width="14"
      height="14"
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      aria-hidden="true"
    >
      <circle cx="12" cy="5" r="3" />
      <line x1="12" y1="22" x2="12" y2="8" />
      <path d="M5 12H2a10 10 0 0 0 20 0h-3" />
    </svg>
  );

  const id = slugify(title);
  const [open, setOpen] = useState(false);
  const [hovered, setHovered] = useState(false);

  useEffect(() => {
    const checkHash = () => {
      if (window.location.hash === `#${id}`) {
        setOpen(true);
      }
    };
    checkHash();
    window.addEventListener("hashchange", checkHash);
    return () => window.removeEventListener("hashchange", checkHash);
  }, [id]);

  useEffect(() => {
    if (!open) return;
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const el = document.getElementById(id);
        if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
      });
    });
  }, [open, id]);

  return (
    <div id={id} className="border rounded-md mb-2">
      <div
        className="relative flex items-center"
        onMouseEnter={() => setHovered(true)}
        onMouseLeave={() => setHovered(false)}
      >
        <a
          href={`#${id}`}
          aria-label="Link to this section"
          style={{
            position: "absolute",
            left: -22,
            opacity: hovered ? 1 : 0,
            transition: "opacity 0.15s",
            color: "#9ca3af",
            textDecoration: "none",
            padding: 4,
          }}
        >
          <AnchorIcon />
        </a>
        <button
          type="button"
          onClick={() => setOpen(!open)}
          className="w-full text-left p-3 font-semibold flex justify-between items-center"
        >
          {title}
          <span>{open ? "−" : "+"}</span>
        </button>
      </div>
      {open && <div className="p-3 pt-0">{children}</div>}
    </div>
  );
};

<DeepLinkAccordion title="Test Section One">
  Lorem ipsum dolor sit amet, placeholder content for the first accordion.
</DeepLinkAccordion>
