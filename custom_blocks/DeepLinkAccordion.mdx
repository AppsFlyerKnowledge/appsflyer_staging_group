import React, { useState, useEffect } from "react";

export const DeepLinkAccordion = ({ title, children }) => {
  const slugify = (text) =>
    (text || "section")
      .toLowerCase()
      .trim()
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/^-+|-+$/g, "") || "section";

  const AnchorIcon = () => (
    <svg
      xmlns="http://www.w3.org/2000/svg"
      viewBox="0 0 576 512"
      width="14"
      height="14"
      fill="currentColor"
      aria-hidden="true"
    >
      <path d="M12.5 222.5L42.2 173c9.4-15.6 29.5-20.6 45.1-11.2l49.5 29.7C151.7 131.9 199.6 96 256 96h32c56.4 0 104.3 35.9 119.2 95.5l49.5-29.7c15.6-9.4 35.7-4.4 45.1 11.2l29.7 49.5c9.4 15.6 4.4 35.7-11.2 45.1l-43.4 26C495 311 496 329.3 496 348c0 106-79.9 193.3-184 205.8V432h40c13.3 0 24-10.7 24-24s-10.7-24-24-24h-40v-64h40c13.3 0 24-10.7 24-24s-10.7-24-24-24h-40v-88c0-17.7-14.3-32-32-32s-32 14.3-32 32v88h-40c-13.3 0-24 10.7-24 24s10.7 24 24 24h40v64h-40c-13.3 0-24 10.7-24 24s10.7 24 24 24h40v121.8C167.9 541.3 88 454 88 348c0-18.7 1-37 8-53.5l-43.4-26c-15.6-9.4-20.6-29.5-11.2-45.1zM288 64a32 32 0 1 0 0-64 32 32 0 1 0 0 64z" />
    </svg>
  );

  const id = slugify(title);
  const [open, setOpen] = useState(false);
  const [hovered, setHovered] = useState(false);

  useEffect(() => {
    const checkHash = () => {
      if (window.location.hash === `#${id}`) {
        setOpen(true);
      }
    };
    checkHash();
    window.addEventListener("hashchange", checkHash);
    return () => window.removeEventListener("hashchange", checkHash);
  }, [id]);

  useEffect(() => {
    if (!open) return;
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const el = document.getElementById(id);
        if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
      });
    });
  }, [open, id]);

  return (
    <div id={id} className="border rounded-md mb-2">
      <div
        className="relative flex items-center"
        onMouseEnter={() => setHovered(true)}
        onMouseLeave={() => setHovered(false)}
      >
        <a
          href={`#${id}`}
          aria-label="Link to this section"
          style={{
            position: "absolute",
            left: -22,
            opacity: hovered ? 1 : 0,
            transition: "opacity 0.15s",
            color: "#9ca3af",
            textDecoration: "none",
            padding: 4,
          }}
        >
          <AnchorIcon />
        </a>
        <button
          type="button"
          onClick={() => setOpen(!open)}
          className="w-full text-left p-3 font-semibold flex justify-between items-center"
        >
          {title}
          <span>{open ? "−" : "+"}</span>
        </button>
      </div>
      {open && <div className="p-3 pt-0">{children}</div>}
    </div>
  );
};

<DeepLinkAccordion title="Test Section One">
  Lorem ipsum dolor sit amet, placeholder content for the first accordion.
</DeepLinkAccordion>
